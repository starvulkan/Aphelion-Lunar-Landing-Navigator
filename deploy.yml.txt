# Builds Aphelion's data from primary sources, checks the ephemeris engine
# against NASA/JPL Horizons, and publishes the site to GitHub Pages.
#
# One-time setup: repository Settings → Pages → Build and deployment →
# Source: "GitHub Actions".

name: Deploy to GitHub Pages

on:
  push:
    branches: ["main", "master"]
  workflow_dispatch:
  schedule:
    # Daily: refresh the JPL Horizons reference table and re-run validation.
    - cron: "17 5 * * *"

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: "pages"
  cancel-in-progress: false

jobs:
  build:
    runs-on: ubuntu-latest
    timeout-minutes: 30
    steps:
      - name: Checkout
        uses: actions/checkout@v5

      - name: Set up Node
        uses: actions/setup-node@v5
        with:
          node-version: "24"

      # LOLA grids (~185 MB) are downloaded once from NASA PDS and cached.
      - name: Cache LOLA downloads
        uses: actions/cache@v5
        with:
          path: .cache/lola
          key: lola-gdr-v1-${{ hashFiles('tools/build-data.mjs', 'js/sites.js') }}
          restore-keys: lola-gdr-v1-

      # Recomputes data/horizons.json and data/dem/* from LOLA, and fetches a
      # fresh 30-day Sun/Earth table from JPL Horizons. If NASA's servers are
      # unreachable the committed data/ folder is used as it is.
      - name: Build data from NASA sources
        id: data
        continue-on-error: true
        run: node tools/build-data.mjs

      - name: Report data build
        if: steps.data.outcome != 'success'
        run: echo "::warning::Data build failed (NASA servers unreachable?). Deploying with the committed data folder."

      # Fails the deploy if the engine disagrees with JPL Horizons by > 0.1°.
      - name: Validate ephemeris against JPL Horizons
        run: node tools/validate.mjs

      # Season analysis must reproduce the team report's Horizons DE441 windows.
      - name: Validate season analysis
        run: node tools/validate-season.mjs

      - name: Check required files
        run: |
          for f in index.html css/styles.css js/app.js js/ephemeris.js js/terrain.js js/sites.js js/viz.js js/scene3d.js js/season.js js/score.js js/learn.js js/lessons.js js/store.js tools/lola-horizon.js data/horizons.json data/patches.i16 data/patches.json vendor/three/three.module.min.js manifest.json graphics/icon.png graphics/logo.png graphics/models/moon_web.glb graphics/models/moon_mobile.glb graphics/textures/earth_bmng_200405_2048.jpg; do
            [ -f "$f" ] || { echo "::error::Missing $f"; exit 1; }
          done

      - name: Assemble site
        run: |
          mkdir -p _site/tools
          cp -r index.html manifest.json css js data graphics vendor DATA_SOURCES.md VERIFY.md LICENSE _site/
          cp tools/lola-horizon.js _site/tools/
          rm -f _site/graphics/README.md
          rm -f _site/graphics/models/moon_small.glb   # original kept in the repo; the app loads moon_web.glb
          touch _site/.nojekyll
          du -sh _site

      - name: Upload artifact
        uses: actions/upload-pages-artifact@v5
        with:
          path: _site

  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - name: Deploy to GitHub Pages
        id: deployment
        uses: actions/deploy-pages@v5
